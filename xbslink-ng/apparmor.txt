#include <tunables/global>

profile xbslink-ng flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities
  capability net_raw,
  capability net_admin,
  capability setgid,
  capability setuid,
  file,
  signal (send) set=(kill,term,int,hup,cont),
  network inet dgram,
  network inet6 dgram,
  network packet raw,

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/local/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Access to options.json
  /data/** rw,

  # Start new profile for xbslink-ng
  /usr/local/bin/xbslink-ng cx -> xbslink_ng,

  profile xbslink_ng flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # Receive signals from S6-Overlay
    signal (receive) peer=*_xbslink-ng,

    # Network capabilities for packet capture and injection
    capability net_raw,
    capability net_admin,
    network inet dgram,
    network inet6 dgram,
    network packet raw,

    # Access to network interfaces
    /sys/class/net/** r,
    /sys/devices/** r,
    /proc/sys/net/** r,

    # Access to BPF for packet filtering
    /dev/bpf* rw,

    # Access to raw sockets
    /dev/net/tun rw,

    # Binary execution
    /usr/local/bin/xbslink-ng r,
    /bin/bash rix,
    /bin/sh rix,

    # Standard files
    /etc/passwd r,
    /etc/group r,
    /etc/nsswitch.conf r,
    /etc/resolv.conf r,
    /dev/tty rw,
    /dev/null rw,
    /dev/zero r,
    /dev/urandom r,

    # Process info
    /proc/*/stat r,
    /proc/*/status r,
    /proc/sys/kernel/random/uuid r,

    # Temporary files
    /tmp/** rwk,
  }
}
